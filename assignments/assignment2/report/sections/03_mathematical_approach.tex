\section{Mathematical Approach}\label{sec:math}
We follow Method~1 of the handout: find the actual kinematic parameters of the robot and use them in place of the nominal ones. Section~\ref{sec:math-problem} states what must be achieved and what is unknown, and Section~\ref{sec:math-model} collects the unknowns as parameters. Section~\ref{sec:math-lsq} estimates them by least squares, Section~\ref{sec:math-init} gives the starting values that the least squares needs, and Section~\ref{sec:math-turn} shows which parameters the data determine. Section~\ref{sec:math-kinematics} gives the improved forward and inverse kinematics. All computations use the exact model of Eq.~\eqref{eq:actual-fk}. Notation follows Section~\ref{sec:variables}.

\subsection{Problem formulation}\label{sec:math-problem}

\subsubsection{Model of the actual robot}
For encoder values $\vct{\theta}$ the physical joint values are $\theta^*_k=\theta_k+\Delta\theta^{\mathrm{rob}}_{k,\mathrm{rnd}}+\Delta\theta^{\mathrm{rob}}_{k,\mathrm{sys}}(\theta_k)$, and the actual end-effector pose relative to the actual base is
\begin{equation}\label{eq:actual-fk}
F^*_{\mathrm{ee}}(\vct{\theta})=F^*_c\,F_{\mathrm{cw}}\,F^*_w\,F_{\mathrm{we}}\Delta F_{\mathrm{we}},
\end{equation}
\begin{equation*}
F^*_c=\Frame\Bigl(I,\ \textstyle\sum_{k=1}^{3}\theta^*_k\vct{d}^*_k\Bigr),\qquad
F^*_w=\Frame\Bigl(\textstyle\prod_{k=4}^{6}R(\vct{d}^*_k,\theta^*_k),\ \vct{0}\Bigr),
\end{equation*}
with $\vct{d}^*_k=R(\vct{\phi}_k)\vct{d}_k$. The wrist rotations are applied in sequence, so $\vct{d}^*_5$ and $\vct{d}^*_6$ are expressed in the frames left by the preceding wrist rotations. The nominal model $F_{\mathrm{ee}}(\vct{\theta})$ is Eq.~\eqref{eq:actual-fk} with $\theta^*_k\to\theta_k$, $\vct{d}^*_k\to\vct{d}_k$, $\Delta F_{\mathrm{we}}\to I$. After a move, $\Delta\theta^{\mathrm{rob}}_{k,\mathrm{rnd}}$ is that of the move; the forward error is evaluated without it.

\subsubsection{Model of the observations}
Sample $j$ is one move to the encoder values $\vct{\theta}_j$ followed by one tracker observation. $mBee$ is fixed to the end-effector and its frame is the end-effector frame (Table~\ref{tab:exact}), so its actual pose relative to the tracker is
\begin{equation}\label{eq:actual-mbee}
F_{\mathrm{tracker}}^{-1}F^*_{\mathrm{rob}}F^*_{\mathrm{ee}}(\vct{\theta}_j)=F^*_X\,F^*_{\mathrm{ee}}(\vct{\theta}_j),
\end{equation}
where $F^*_{\mathrm{ee}}$ contains the random joint error of move $j$. Its markers are actually at
\begin{equation}\label{eq:actual-markers}
\vct{b}^*_{k,j}=F^*_X\,F^*_{\mathrm{ee}}(\vct{\theta}_j)\,\vct{a}^*_k .
\end{equation}
The tracker reports positions $\vct{b}_{k,j}$ related to them by (handout p.~2)
\begin{equation}\label{eq:reported-markers}
\vct{b}^*_{k,j}=\vct{b}_{k,j}+\vct{\epsilon}^{\mathrm{trk}}_{\mathrm{sys}}(\vct{b}_{k,j})+\vct{\epsilon}_{b,k,j},
\end{equation}
where $\vct{\epsilon}_{b,k,j}$ is new in every observation and $\vct{\epsilon}^{\mathrm{trk}}_{\mathrm{sys}}$ is a fixed function of position. $\Fun{SampleMarkerBodies}$ then returns the pose that best maps the known markers onto the reported positions, the observation $\hat F_{m,j}$ (the handout's $F_m$ with $m=mBee$; Table~\ref{tab:symbols}):
\begin{equation}\label{eq:observation}
\hat F_{m,j}=\arg\min_{F}\sum_k\norm{F\vct{a}^*_k-\vct{b}_{k,j}}^2 .
\end{equation}
Because of the tracker errors, the actual pose of Eq.~\eqref{eq:actual-mbee} differs from the observation by a small frame,
\begin{equation}\label{eq:reading-error}
F^*_X\,F^*_{\mathrm{ee}}(\vct{\theta}_j)=\hat F_{m,j}\,\Delta F_{m,j},\qquad\Delta F_{m,j}=\Frame\bigl(I+\sk(\vct{\alpha}_{m,j}),\vct{\epsilon}_{m,j}\bigr),
\end{equation}
with $\Delta F_{m,j}=I$ when $\vct{\epsilon}^{\mathrm{trk}}_{\mathrm{sys}}=\vct{\epsilon}_b=\vct{0}$. $\Delta F_{m,j}$ is unknown. The calibration below states each relation for the actual poses and applies it to the observations with $\Delta F_{m,j}$ written out as the error of the observation.

\subsection{Calibration model}\label{sec:math-model}
\subsubsection{Joint distortion}
Each $\Delta\theta^{\mathrm{rob}}_{k,\mathrm{sys}}$ is stored as its values $c_{k,l}=\Delta\theta^{\mathrm{rob}}_{k,\mathrm{sys}}(\tau_{k,l})$ at chosen joint values (knots) $\tau_{k,0}<\dots<\tau_{k,L_k}$ that include both ends of the joint range, and is read between knots by straight-line interpolation:
\begin{equation}\label{eq:knots}
\Delta\hat\theta^{\mathrm{rob}}_{k,\mathrm{sys}}(\theta)=(1-\lambda)\,c_{k,l}+\lambda\,c_{k,l+1},\qquad\lambda=\frac{\theta-\tau_{k,l}}{\tau_{k,l+1}-\tau_{k,l}}\ \text{for}\ \tau_{k,l}\leq\theta\leq\tau_{k,l+1}.
\end{equation}
This form needs no assumption beyond the handout's bounds: between knots its error is at most $\tfrac12\omega^{\mathrm{rob}}_{k,\mathrm{sys}}(\tau_{k,l+1}-\tau_{k,l})$.

\subsection{Estimation by least squares}\label{sec:math-lsq}
The parameters are estimated by fitting the exact model of Eq.~\eqref{eq:actual-fk} to all observations at once. For a parameter vector $\vct{x}$, sample $j$ is predicted at the pose $F_X(\vct{x})\hat F_{\mathrm{ee}}(\vct{\theta}_j;\vct{x})$; its difference from the observation,
\begin{equation}\label{eq:residual}
\Frame\bigl(I+\sk(\vct{\alpha}_j),\vct{\epsilon}_j\bigr)=\hat F_{m,j}^{-1}\,F_X(\vct{x})\hat F_{\mathrm{ee}}(\vct{\theta}_j;\vct{x}),\qquad\vct{r}_j(\vct{x})=[\vct{\alpha}_j;\vct{\epsilon}_j],
\end{equation}
is the residual of the sample. At the actual parameters it is exactly the observation error $\Delta F_{m,j}$ of Eq.~\eqref{eq:reading-error}. The estimate makes the residuals as small as possible:
\begin{equation}\label{eq:lsq}
\hat{\vct{x}}=\arg\min_{\vct{x}}\sum_j\norm{W_j\,\vct{r}_j(\vct{x})}^2,
\end{equation}
where the weight $W_j$ scales rotation and translation by their expected measurement errors. Let $\vct{r}(\vct{x})$ be all weighted residuals stacked into one vector. Since $\vct{r}$ is nonlinear in $\vct{x}$, the minimum is found by Gauss--Newton steps from a starting value. Near the current $\vct{x}$,
\begin{equation*}
\vct{r}(\vct{x}+\vct{\delta})\approx\vct{r}(\vct{x})+J\vct{\delta},\qquad J=\frac{\partial\vct{r}}{\partial\vct{x}},
\end{equation*}
where each column of $J$ tells how all residuals change when one parameter changes. The step minimizing $\norm{\vct{r}+J\vct{\delta}}^2$ solves
\begin{equation}\label{eq:gauss-newton}
J^TJ\,\vct{\delta}=-J^T\vct{r},
\end{equation}
and $\vct{x}\leftarrow\vct{x}+\vct{\delta}$ is repeated until $\vct{\delta}$ is negligible. Two conditions follow, each treated in its own subsection:
\begin{itemize}
\item Gauss--Newton converges only from a start close to the minimum, while the nominal model can be off by tens of millimeters (Table~\ref{tab:unknowns}). Section~\ref{sec:math-init} derives starting values.
\item Eq.~\eqref{eq:gauss-newton} has a unique solution only if $J^TJ$ is invertible, i.e.\ if $J$ has full column rank. If some direction $\vct{v}$ of parameter change leaves all residuals unchanged, $J\vct{v}=\vct{0}$, then $\vct{\delta}$ and $\vct{\delta}+s\vct{v}$ fit equally well for every $s$, and the parameters are not determined. Section~\ref{sec:math-turn} shows that the parameter vector of Eq.~\eqref{eq:params} has exactly one such direction and removes it.
\end{itemize}
The fit uses any samples, including samples in which several joints move together.

\subsection{Starting values from single-joint motions}\label{sec:math-init}
A start close to the solution comes from the structure of Eq.~\eqref{eq:actual-fk}: every unknown belongs to one joint or to a fixed frame, and no term couples two joints. Each joint can therefore be estimated from motions of that joint alone, and the base and the tool afterwards: each wrist joint from orientations (Step~A), each stage from positions (Step~B), and the base and the tool from these results (Step~C). Each step fits its relation by least squares over many observations, so that the observation errors are averaged. The steps use each observation for one purpose only and neglect small terms; the fit of Section~\ref{sec:math-lsq} removes both.

\subsubsection{Relative motion of one joint}\label{sec:math-single}
A single-joint motion is compared with a sample at a fixed configuration, so that everything that does not move cancels. Let $j=0$ be a sample at a base configuration with encoder values $\vct{\theta}_0$, and $j$ a sample in which joint $k$ alone differs from $\vct{\theta}_0$. Every move draws a new random error for all six joints, so the joints held at their base values are not exactly where they were in sample $0$; the difference is of the order of $\sigma_{k,\mathrm{rnd}}$ ($0.005$\,mm or $10^{-4}$\,rad). Steps~A--C neglect it; the fit of Section~\ref{sec:math-lsq} treats it as noise.

Frames are written as homogeneous matrices, points and directions as homogeneous vectors:
\begin{equation*}
F=\begin{bmatrix}R&\vct{p}\\\vct{0}^T&1\end{bmatrix},\qquad
\tilde{\vct{p}}=\begin{bmatrix}\vct{p}\\1\end{bmatrix}\ \text{(point)},\qquad
\tilde{\vct{d}}=\begin{bmatrix}\vct{d}\\0\end{bmatrix}\ \text{(direction)},
\end{equation*}
so that a frame moves a point and only turns a direction. Two kinds of pose appear in the derivation:
\begin{itemize}
\item the \emph{actual} pose of $mBee$ in sample $j$, $F^*_{m,j}=F^*_XF^*_{\mathrm{ee}}(\vct{\theta}_j)$, which is hidden;
\item the \emph{observation} $\hat F_{m,j}$, the frame that the tracker computes from the reported markers by SVD registration, Eq.~\eqref{eq:observation}; it is an estimate of $F^*_{m,j}$, hence the hat.
\end{itemize}
They are related by Eq.~\eqref{eq:reading-error}, $F^*_{m,j}=\hat F_{m,j}\,\Delta F_{m,j}$. The motion from sample $0$ to sample $j$ is described by the relative frame, actual and read:
\begin{equation}\label{eq:rel-frames}
T^*_j=F^*_{m,j}\bigl(F^*_{m,0}\bigr)^{-1},\qquad \hat F_{m,j}\hat F_{m,0}^{-1}=T^*_j\;F^*_{m,0}\,\Delta F_{m,j}^{-1}\Delta F_{m,0}\,\bigl(F^*_{m,0}\bigr)^{-1}.
\end{equation}
$T^*_j$ is what the robot actually did; $\hat F_{m,j}\hat F_{m,0}^{-1}$ is computed from two observations and differs from $T^*_j$ by the last factor, which is made of the two observation errors and is $I$ when they are equal. The relations below are derived for $T^*_j$; Steps~A--C apply them to the observed relative frame, which equals $T^*_j$ up to the observation errors.

By Eq.~\eqref{eq:actual-fk} the actual pose is the product
\begin{equation}\label{eq:actual-parts}
F^*_{m,j}=F^*_X\,F^*_{c,j}\,F_{\mathrm{cw}}\,F^*_{w,j}\,F_{\mathrm{we}}\Delta F_{\mathrm{we}},\qquad
F^*_{c,j}=\begin{bmatrix}I&\sum_{i=1}^{3}\theta^*_{i,j}\vct{d}^*_i\\\vct{0}^T&1\end{bmatrix} .
\end{equation}
$F^*_{c,j}$ has the rotation block $I$, so the stages change only the position of $mBee$, not its orientation; this is the handout's remark that the first three joints are decoupled from the last three.

If joint $k$ is a stage, the other factors are the same in both samples and $F^*_{c,j}=F^*_{c,0}\,S$ with the pure translation
\begin{equation*}
S=\begin{bmatrix}I&(\theta^*_{k,j}-\theta^*_{k,0})\,\vct{d}^*_k\\\vct{0}^T&1\end{bmatrix}.
\end{equation*} Translations commute, so $F^*_{m,j}=F^*_X\,S\,(F^*_X)^{-1}F^*_{m,0}$ and
\begin{equation}\label{eq:stage-line}
T^*_j=F^*_X\,S\,(F^*_X)^{-1}=\begin{bmatrix}I&(\theta^*_{k,j}-\theta^*_{k,0})\,\vct{\ell}_k\\\vct{0}^T&1\end{bmatrix},\qquad\tilde{\vct{\ell}}_k=F^*_X\,\tilde{\vct{d}}^*_k :
\end{equation}
$mBee$ is translated along $\vct{\ell}_k$, the stage direction in the tracker frame, by the physical travel.

If joint $k$ is a wrist joint, write $F^*_{w,j}=F_{A,k}\,F_R(\vct{d}^*_k,\theta^*_{k,j})\,F_{B,k}$, where $F_R(\vct{d},\theta)$ is the rotation about $\vct{d}$ by $\theta$ as a frame and $F_{A,k}$, $F_{B,k}$ are the rotations of the wrist joints before and after joint $k$ ($F_{A,4}=F_{B,6}=I$); both are the same in the two samples. With $G_k=F^*_XF^*_{c,0}F_{\mathrm{cw}}F_{A,k}$, the frame of joint $k$ in the tracker frame at the base configuration, $F^*_{m,j}=G_k\,F_R(\vct{d}^*_k,\theta^*_{k,j})\,F_{B,k}F_{\mathrm{we}}\Delta F_{\mathrm{we}}$, and everything after the joint cancels:
\begin{equation}\label{eq:wrist-turn}
T^*_j=G_k\,F_R\bigl(\vct{d}^*_k,\theta^*_{k,j}-\theta^*_{k,0}\bigr)\,G_k^{-1},\qquad\tilde{\vct{\ell}}_k=G_k\,\tilde{\vct{d}}^*_k,\qquad\tilde{\vct{o}}_w=G_k\begin{bmatrix}\vct{0}\\1\end{bmatrix} .
\end{equation}
This is the rotation by the physical travel $\theta^*_{k,j}-\theta^*_{k,0}$ about the line through the wrist center $\vct{o}_w$ (the origin of $G_k$) with direction $\vct{\ell}_k$, the joint axis in the tracker frame. In particular $T^*_j\tilde{\vct{o}}_w=\tilde{\vct{o}}_w$: the wrist center does not move.

In both cases the relative frame contains only the joint's axis, its travel and, for a wrist joint, the wrist center; the base translation and the tool have cancelled.

\subsubsection{Step A: wrist joints}
For wrist joint $k$, the observed relative frame is, by Eqs.~\eqref{eq:rel-frames} and \eqref{eq:wrist-turn},
\begin{equation}\label{eq:stepA}
\hat F_{m,j}\hat F_{m,0}^{-1}=G_k\,F_R\bigl(\vct{d}^*_k,\theta^*_{k,j}-\theta^*_{k,0}\bigr)\,G_k^{-1}\quad\text{up to the observation errors.}
\end{equation}
Its rotation block is the turn about $\vct{\ell}_k$ by the travel. Written as an axis and an angle, with the sign of the axis chosen along the positive direction of joint $k$, it gives the axis $\vct{\ell}_k$ and the angle
\begin{equation*}
\psi_{k,j}=\theta^*_{k,j}-\theta^*_{k,0},
\end{equation*}
both up to the observation errors.

\subsubsection{Step B: stages}
For stage $k$, by Eqs.~\eqref{eq:rel-frames} and \eqref{eq:stage-line},
\begin{equation}\label{eq:stepB}
\hat F_{m,j}\hat F_{m,0}^{-1}=\begin{bmatrix}I&(\theta^*_{k,j}-\theta^*_{k,0})\,\vct{\ell}_k\\\vct{0}^T&1\end{bmatrix}\quad\text{up to the observation errors:}
\end{equation}
the observed relative frames are translations along one line. The direction of the line fitted to their translations is $\vct{\ell}_k$, and since $\vct{\ell}_k$ is a unit vector, the length of each translation along it is the travel $\theta^*_{k,j}-\theta^*_{k,0}$, written $\lambda_{k,j}$.

\subsubsection{From travel to distortion}\label{sec:math-travel}
Steps~A and~B give the travel of joint $k$ from the base configuration, $\psi_{k,j}$ or $\lambda_{k,j}$, i.e.\ $\theta^*_{k,j}-\theta^*_{k,0}$. By the joint model of Section~\ref{sec:math-problem},
\begin{equation*}
\theta^*_{k,j}=\theta_{k,j}+\Delta\theta^{\mathrm{rob}}_{k,\mathrm{rnd},j}+\Delta\theta^{\mathrm{rob}}_{k,\mathrm{sys}}(\theta_{k,j}),
\end{equation*}
so eliminating $\theta^*_{k,j}$ gives the distortion at the commanded value:
\begin{equation}\label{eq:stepA-y}
y_{k,j}\equiv\bigl(\theta^*_{k,j}-\theta^*_{k,0}\bigr)+\theta^*_{k,0}-\theta_{k,j}=\Delta\theta^{\mathrm{rob}}_{k,\mathrm{sys}}(\theta_{k,j})+\Delta\theta^{\mathrm{rob}}_{k,\mathrm{rnd},j},
\end{equation}
up to the observation errors. Each sample gives one measurement $y_{k,j}$ of the distortion, disturbed by the random joint error of its move and by the observation errors. It needs the physical angle $\theta^*_{k,0}$ at the base configuration, an absolute value that the tracker cannot see, since it measures only travels. Homing provides it.

\subsubsection{Absolute reference from homing}
$\Fun{HomeRobot}$ stops at $\theta^*_k=0$ and returns $\theta_{\mathrm{encoders},k}=-\Delta\theta^{\mathrm{rob}}_{k,\mathrm{rnd}}-\Delta\theta^{\mathrm{rob}}_{k,\mathrm{sys}}(\theta_{\mathrm{encoders},k})$. Each homing draws a new $\Delta\theta^{\mathrm{rob}}_{k,\mathrm{rnd}}$, so the returned values scatter by about $\sigma_{k,\mathrm{rnd}}$, over which $\Delta\theta^{\mathrm{rob}}_{k,\mathrm{sys}}$ changes by at most $\omega^{\mathrm{rob}}_{k,\mathrm{sys}}\sigma_{k,\mathrm{rnd}}$. The average of several homings therefore gives one value of each distortion function without the tracker:
\begin{equation}\label{eq:home-knot}
\Delta\theta^{\mathrm{rob}}_{k,\mathrm{sys}}\bigl(\bar\theta_{\mathrm{encoders},k}\bigr)=-\bar\theta_{\mathrm{encoders},k}.
\end{equation}
This is an absolute value of $\Delta\theta^{\mathrm{rob}}_{k,\mathrm{sys}}$. Together with the measurements $y_{k,j}$ it determines the knot values of Eq.~\eqref{eq:knots}; how it enters the knots is part of the algorithm. How it gives $\theta^*_{k,0}$ depends on the base configuration.

\subsubsection{Base configuration of the single-joint motions}\label{sec:math-base}
Two choices of the base configuration differ in how $\theta^*_{k,0}$ is obtained.
\begin{itemize}
\item \emph{Physical home}, the configuration reached by $\Fun{HomeRobot}$: then $\theta^*_{k,0}=0$ and $F_{A,k}=I$, so the observations give $\theta^*_{k,j}$ itself and $\tilde{\vct{\ell}}_k=F^*_X\tilde{\vct{d}}^*_k$.
\item \emph{Another configuration}, for example $\vct{\theta}_0=\vct{0}$. The physical angle at the base contains the distortion there,
\begin{equation*}
\theta^*_{k,0}=\theta_{k,0}+\Delta\theta^{\mathrm{rob}}_{k,\mathrm{sys}}(\theta_{k,0}),
\end{equation*}
which is not measured. Replacing it by the homing value of Eq.~\eqref{eq:home-knot} errs, by the slope bound, by at most
\begin{equation}\label{eq:base-error}
\bigl|\Delta\theta^{\mathrm{rob}}_{k,\mathrm{sys}}(\theta_{k,0})-\Delta\theta^{\mathrm{rob}}_{k,\mathrm{sys}}(\bar\theta_{\mathrm{encoders},k})\bigr|\leq\omega^{\mathrm{rob}}_{k,\mathrm{sys}}\,\bigl|\theta_{k,0}-\bar\theta_{\mathrm{encoders},k}\bigr| .
\end{equation}
For $\vct{\theta}_0=\vct{0}$, where $|\bar\theta_{\mathrm{encoders},k}|\leq\delta^{\mathrm{rob}}_{k,\mathrm{sys}}$, this is at most
\begin{equation*}
\omega^{\mathrm{rob}}_{k,\mathrm{sys}}\,\delta^{\mathrm{rob}}_{k,\mathrm{sys}}=
\begin{cases}5\times10^{-4}\cdot0.2\,\mathrm{mm}=10^{-4}\,\mathrm{mm}&\text{(stage)},\\5\times10^{-5}\cdot0.01\,\mathrm{rad}=5\times10^{-7}\,\mathrm{rad}&\text{(wrist joint)}.\end{cases}
\end{equation*}
In addition $F_{A,k}\neq I$: the axes of joints 5 and 6 are seen turned by the base angles of the preceding wrist joints (up to $0.01$\,rad), which Step~C must take into account.
\end{itemize}
Both choices are admissible; the choice belongs to the algorithm.

\subsubsection{Step C: base and tool}
The wrist center does not move during a wrist motion (Eq.~\eqref{eq:wrist-turn}), so every wrist sample satisfies $T^*_j\tilde{\vct{o}}_w=\tilde{\vct{o}}_w$. With the observed frames this holds up to the observation errors, and $\vct{o}_w$ is found by linear least squares over all wrist samples:
\begin{equation}\label{eq:pivot}
\hat{\vct{o}}_w=\arg\min_{\vct{o}}\sum_j\norm{\hat F_{m,j}\hat F_{m,0}^{-1}\begin{bmatrix}\vct{o}\\1\end{bmatrix}-\begin{bmatrix}\vct{o}\\1\end{bmatrix}}^2 ,
\end{equation}
the same problem as pivot calibration: one point that all the motions leave in place. The base frame $\hat F_X$ then follows from three conditions on directions and points, and the tool from the observation at the base:
\begin{itemize}
\item orientation: $\hat F_X\tilde{\vct{d}}_1=\tilde{\vct{\ell}}_1$, since $\vct{d}^*_1=\vct{d}_1$, which fixes two of its three rotational degrees of freedom; for the turn about $\vct{x}$ we require $\hat F_X$ to map the $xy$ plane of the base onto the plane spanned by $\vct{\ell}_1$ and $\vct{\ell}_2$, i.e.\ we place $\vct{d}^*_2$ in the $xy$ plane of the base. Section~\ref{sec:math-turn} shows that the data cannot determine this turn better;
\item position: at the base configuration the wrist center lies at $\sum_i\theta^*_{i,0}\hat{\vct{d}}_i+[500,0,0]^T$ in the base, so $\hat F_X$ must map this point onto $\hat{\vct{o}}_w$;
\item axes and tilts: $\tilde{\hat{\vct{d}}}_k=\hat F_X^{-1}\tilde{\vct{\ell}}_k$ for the stages and $(\hat F_X\hat F_{A,k})^{-1}\tilde{\vct{\ell}}_k$ for the wrist joints, with $\hat F_{A,k}$ the preceding wrist rotations at the base ($I$ at the physical home); then $\hat{\vct{\phi}}_k\approx\vct{d}_k\times\hat{\vct{d}}_k$;
\item tool: by Eq.~\eqref{eq:actual-parts} at the base, $F^*_{m,0}=F^*_XF^*_{c,0}F_{\mathrm{cw}}F^*_{w,0}F_{\mathrm{we}}\Delta F_{\mathrm{we}}$, so
\begin{equation*}
\Delta\hat F_{\mathrm{we}}=\bigl(\hat F_X\hat F_{c,0}F_{\mathrm{cw}}\hat F_{w,0}F_{\mathrm{we}}\bigr)^{-1}\hat F_{m,0},
\end{equation*}
with the observation $\hat F_{m,0}$ in place of the actual pose.
\end{itemize}

\subsubsection{Angles between axes}
Steps~A and~B give all six actual axes $\vct{\ell}_1,\dots,\vct{\ell}_6$ in the tracker frame. The angles between them do not depend on the base (hint~1(c) of the handout).

\subsection{Which parameters the data determine}\label{sec:math-turn}
A direction $\vct{v}$ with $J\vct{v}=\vct{0}$ exists if two different parameter vectors predict the same pose for every sample. By Eq.~\eqref{eq:reading-error}, the actual pose $F^*_XF^*_{\mathrm{ee}}(\vct{\theta})$ differs from an observation only by the observation error, so the tracker sees only this product, never the base $F^*_X$ and the chain $F^*_{\mathrm{ee}}(\vct{\theta})$ separately: two models that give the same product for every $\vct{\theta}$ produce the same observations, and homing distinguishes them only if they assign different physical joint angles to the same encoder values.

\subsubsection{The turn of the base about the first axis}
The rotation of the base $F^*_X$ has three degrees of freedom. The single-joint motions give $\tilde{\vct{\ell}}_1=F^*_X\tilde{\vct{d}}_1$ with $\vct{d}_1=\vct{x}$ exact, which fixes two of them. The third, the turn of the base about $\vct{x}$, would follow from $\tilde{\vct{\ell}}_2=F^*_X\tilde{\vct{d}}^*_2$ if $\vct{d}^*_2$ were known, but $\vct{d}^*_2$ is tilted by the unknown $\vct{\phi}_2$. The question is whether a turn of the base about $\vct{x}$ can be traded against the tilts.

Describe the same robot with the base turned by $c$ about $\vct{x}$ and, relative to this base, every axis and the tool turned back by $-c$: the robot in the tracker frame is unchanged. The observations cannot tell the two descriptions apart. In formulas, with $Q(c)=F_R(\vct{x},c)$ the turn by $c$ about $\vct{x}$ as a frame, this is the insertion of $Q(c)\,Q(-c)=I$ after the base:
\begin{equation}\label{eq:turn-insert}
F^*_XF^*_{\mathrm{ee}}(\vct{\theta})=\bigl(F^*_XQ(c)\bigr)\bigl(Q(-c)\,F^*_{\mathrm{ee}}(\vct{\theta})\bigr).
\end{equation}
The first factor is the turned base. It remains to show that the second factor is again a robot of the form of Eq.~\eqref{eq:actual-fk}. A rotation frame $Q$ can be moved to the right past a translation frame, which turns its translation by $Q$ and leaves a translation along the axis of $Q$ unchanged, and past a rotation frame $F_R(\vct{d},\theta)$, which turns its axis by $Q$ and keeps its angle. With $Q(-c)$ and the chain of Eq.~\eqref{eq:actual-fk}:
\begin{align*}
Q(-c)\,F^*_{\mathrm{ee}}(\vct{\theta})
&=Q(-c)\,F^*_c\,F_{\mathrm{cw}}\,F^*_w\,F_{\mathrm{we}}\,\Delta F_{\mathrm{we}}\\
&=F'_c\,Q(-c)\,F_{\mathrm{cw}}\,F^*_w\,F_{\mathrm{we}}\,\Delta F_{\mathrm{we}}
&&F'_c:\ \text{axes }\tilde{\vct{d}}'_i=Q(-c)\tilde{\vct{d}}^*_i\\
&=F'_c\,F_{\mathrm{cw}}\,Q(-c)\,F^*_w\,F_{\mathrm{we}}\,\Delta F_{\mathrm{we}}
&&[500,0,0]^T\text{ lies along }\vct{x}\\
&=F'_c\,F_{\mathrm{cw}}\,F'_w\,Q(-c)\,F_{\mathrm{we}}\,\Delta F_{\mathrm{we}}
&&F'_w:\ \text{axes }\tilde{\vct{d}}'_k=Q(-c)\tilde{\vct{d}}^*_k\\
&=F'_c\,F_{\mathrm{cw}}\,F'_w\,F_{\mathrm{we}}\,Q(-c)\,\Delta F_{\mathrm{we}}
&&[100,0,0]^T\text{ lies along }\vct{x}.
\end{align*}
The result is again the chain of Eq.~\eqref{eq:actual-fk}, with the same joint angles, the axes $\tilde{\vct{d}}'_i=Q(-c)\tilde{\vct{d}}^*_i$ and the tool error $\Delta F'_{\mathrm{we}}=Q(-c)\Delta F_{\mathrm{we}}$; call it $F'_{\mathrm{ee}}$. It is an admissible model, since a turn about $\vct{x}$ leaves $\vct{d}_1=\vct{x}$ unchanged. By Eq.~\eqref{eq:turn-insert},
\begin{equation}\label{eq:turn-chain}
F^*_XF^*_{\mathrm{ee}}(\vct{\theta})=\bigl(F^*_XQ(c)\bigr)\,F'_{\mathrm{ee}}(\vct{\theta})\quad\text{for every }\vct{\theta},
\end{equation}
so the turned base, axes and tool give the same observations and the same joint angles: neither the tracker nor homing determines $c$.

The turn must be about $\vct{x}$: a turn about any other direction would change $\vct{d}_1$, which has no tilt parameter, and would not commute with the translations of $F_{\mathrm{cw}}$ and $F_{\mathrm{we}}$, which both lie along $\vct{x}$. In the parameters of Table~\ref{tab:params}, the turn changes $\vct{\alpha}_X$, the tilts of axes 2, 3, 5, 6 and $\vct{\alpha}_{\mathrm{we}}$ together; this combination is the direction $\vct{v}$ with $J\vct{v}=\vct{0}$.

\subsubsection{Directions removed by homing}
The same construction gives other changes that leave all products $F^*_XF^*_{\mathrm{ee}}(\vct{\theta})$ unchanged: a shift $\vct{s}$ of the base compensated by the stages, with every $\Delta\theta^{\mathrm{rob}}_{i,\mathrm{sys}}$, $i\leq3$, shifted by the constant $-\vct{d}_i\cdot\vct{s}$, or a turn inserted between two wrist joints, compensated by a constant shift of a wrist joint's distortion. Each requires a constant shift of a whole distortion function, and so changes the physical joint angle that the model assigns to each encoder value. Homing measures an absolute value of every distortion function (Eq.~\eqref{eq:home-knot}), which forbids such shifts. Without homing these would be further directions with $J\vct{v}=\vct{0}$; with it, the turn about $\vct{x}$ is the only one, since it leaves every physical joint angle unchanged.

\subsubsection{Removing the turn and choosing it afterwards}
The direction is removed by fixing one parameter along it, a convention:
\begin{equation}\label{eq:turn-convention}
\vct{\phi}_2\cdot\vct{x}=0,
\end{equation}
i.e.\ $\vct{d}^*_2$ lies in the $xy$ plane of the base, as already used in Step~C. The tilt of axis 2 then has one component, the parameter vector has $9$ tilt components, and $J$ has full column rank. After the fit, the turn is chosen from the only information about it. After a turn by $c$ the estimated model has the base $\hat F_XQ(c)$, the axes and the tool error
\begin{equation*}
\tilde{\hat{\vct{d}}}_k(c)=Q(-c)\,\tilde{\hat{\vct{d}}}_k,\qquad\Delta\hat F_{\mathrm{we}}(c)=Q(-c)\,\Delta\hat F_{\mathrm{we}},
\end{equation*}
and the base placement error relative to the nominal base $F_{\mathrm{rob}}$ becomes
\begin{equation*}
\Delta F_{\mathrm{rob}}(c)=F_{\mathrm{tracker}}\,\hat F_X\,Q(c)\,F_{\mathrm{rob}}^{-1}.
\end{equation*} Let $\norm{\vct{\phi}_k(c)}$ be the angle between $\hat{\vct{d}}_k(c)$ and $\vct{d}_k$, and $\norm{\vct{\alpha}_{\mathrm{we}}(c)}$, $\norm{\vct{\alpha}_{\mathrm{rob}}(c)}$ the rotation angles of $\Delta\hat F_{\mathrm{we}}(c)$ and $\Delta F_{\mathrm{rob}}(c)$. The data give no information about $c$; the only information is the handout's bounds, which the actual robot satisfies:
\begin{equation}\label{eq:turn-bounds}
\norm{\vct{\phi}_k(c)}\leq\phi_{\max}\ (k=2,3,5,6),\qquad\norm{\vct{\alpha}_{\mathrm{we}}(c)}\leq0.01,\qquad\norm{\vct{\alpha}_{\mathrm{rob}}(c)}\leq\alpha^{\max}_{\mathrm{rob}} .
\end{equation}
The tilts of $\vct{d}_1$ and $\vct{d}_4$ are not listed because both nominal axes lie along $\vct{x}$, and a turn about $\vct{x}$ does not change their angle to $\vct{x}$. The values of $c$ that satisfy Eq.~\eqref{eq:turn-bounds} form an interval $[c_{\mathrm{lo}},c_{\mathrm{hi}}]$ that contains the actual value. We take its midpoint,
\begin{equation}\label{eq:turn-choice}
\hat c=\tfrac12\bigl(c_{\mathrm{lo}}+c_{\mathrm{hi}}\bigr),\qquad\bigl|\hat c-c^*\bigr|\leq\tfrac12\bigl(c_{\mathrm{hi}}-c_{\mathrm{lo}}\bigr),
\end{equation}
which gives the smallest worst-case error of $c$.

By Eqs.~\eqref{eq:turn-insert} and \eqref{eq:turn-chain}, $c$ has the following effects:
\begin{itemize}
\item predicted poses in the tracker frame, $F_X\hat F_{\mathrm{ee}}(\vct{\theta})$, do not depend on $c$;
\item the base-relative pose $\hat F_{\mathrm{ee}}(\vct{\theta})$ becomes $Q(-c)\hat F_{\mathrm{ee}}(\vct{\theta})$, and the individual tilts change with $c$;
\item the angles between axes do not depend on $c$, since $\vct{d}'_i\cdot\vct{d}'_l=\vct{d}^*_i\cdot\vct{d}^*_l$; they are the quantities to compare with the ground truth.
\end{itemize}

\subsection{Improved kinematics}\label{sec:math-kinematics}
\subsubsection{Forward kinematics}
$\Fun{ImprovedFwdKins}(\vct{\theta})=\hat F_{\mathrm{ee}}(\vct{\theta})$ is Eq.~\eqref{eq:actual-fk} with the estimated parameters: $\hat\theta_k=\theta_k+\Delta\hat\theta^{\mathrm{rob}}_{k,\mathrm{sys}}(\theta_k)$ from Eq.~\eqref{eq:knots}, $\hat{\vct{d}}_k=R(\hat{\vct{\phi}}_k)\vct{d}_k$, and $F_{\mathrm{we}}\Delta\hat F_{\mathrm{we}}$.

\subsubsection{Inverse kinematics}
For a target $F$ relative to the base we need $\vct{\theta}$ with $\hat F_{\mathrm{ee}}(\vct{\theta})=F$, i.e.\ $\vct{g}(\vct{\theta})=\vct{0}$, where $\vct{g}=[\vct{\alpha};\vct{\epsilon}]$ of $\hat F_{\mathrm{ee}}(\vct{\theta})^{-1}F$. With tilted axes there is no direct formula, so we start from the nominal solution, $\vct{\theta}^{(0)}=\Fun{InvKins}(F)$, and correct it repeatedly (Newton's method):
\begin{equation}\label{eq:ik-iter}
\vct{\theta}^{(i+1)}=\vct{\theta}^{(i)}+J(\vct{\theta}^{(i)})^{-1}\vct{g}(\vct{\theta}^{(i)}),\qquad J=\partial\vct{g}/\partial\vct{\theta}.
\end{equation}
The start differs from the solution only by the calibration corrections, which are small (Table~\ref{tab:unknowns}), so a few steps suffice. Near $\cos\theta_5=0$, where the axes of joints 4 and 6 line up, $J$ is nearly singular and $\theta_4$, $\theta_6$ are poorly determined; limiting and damping the steps (Levenberg--Marquardt), as the provided \path{inv_kins} does, keeps the iteration stable there. Each of the two nominal wrist branches is refined, and the one within the joint limits nearest the current configuration is returned.
